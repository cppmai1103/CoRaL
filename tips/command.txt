1. Check how many avaible GPU?
sinfo -N -h -p long,cscc-gpu-p \
  -O NodeList:15,StateCompact:12,Gres:60,GresUsed:80 |
awk '
BEGIN {
    printf "%-12s %-10s %-10s %-10s\n",
           "NODE", "STATE", "TOTAL", "FREE"
}
{
    split($3, total, ":")
    split($4, used, ":")
    free = (total[3]+0) - (used[3]+0)

    if ($2 ~ /down|drain|fail|maint/ || $2 ~ /\*/)
        free = 0

    if (free > 0)
        printf "%-12s %-10s %-10d %-10d\n",
               $1, $2, total[3]+0, free
}'
